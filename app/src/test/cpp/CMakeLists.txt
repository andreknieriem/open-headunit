cmake_minimum_required(VERSION 3.22)
project(audio_callback_tests LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
add_executable(callback_pcm_buffer_test callback_pcm_buffer_test.cpp)
target_compile_features(callback_pcm_buffer_test PRIVATE cxx_std_17)
target_link_libraries(callback_pcm_buffer_test PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(callback_pcm_buffer_test PRIVATE /W4 /WX)
else()
    target_compile_options(callback_pcm_buffer_test PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME callback_pcm_buffer COMMAND callback_pcm_buffer_test)
